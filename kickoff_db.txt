CREATE DATABASE kickoff_db;

USE kickoff_db;

-- 1. Admin Table
CREATE TABLE Admin (
    adminID INT AUTO_INCREMENT PRIMARY KEY,
    username VARCHAR(100) NOT NULL,
    password VARCHAR(255) NOT NULL
);

-- 2. Team Table
CREATE TABLE Team (
    teamID INT AUTO_INCREMENT PRIMARY KEY,
    teamName VARCHAR(100) NOT NULL,
    teamIcon VARCHAR(255),
    adminID INT,
    FOREIGN KEY (adminID) REFERENCES Admin(adminID)
);

-- 3. Fixture Table (Added ON DELETE CASCADE for foreign keys referencing Team)
CREATE TABLE Fixture (
    fixtureID INT AUTO_INCREMENT PRIMARY KEY,
    homeTeamID INT,
    awayTeamID INT,
    matchDate DATE,
    matchTime TIME,
    venue VARCHAR(150),
    matchday INT,
    isMatchOfWeek TINYINT,
    adminID INT,
    FOREIGN KEY (homeTeamID) REFERENCES Team(teamID) ON DELETE CASCADE,
    FOREIGN KEY (awayTeamID) REFERENCES Team(teamID) ON DELETE CASCADE,
    FOREIGN KEY (adminID) REFERENCES Admin(adminID)
);

-- 4. NewsArticle Table
CREATE TABLE NewsArticle (
    newsID INT AUTO_INCREMENT PRIMARY KEY,
    title VARCHAR(250) NOT NULL,
    content TEXT,
    publishDate DATE,
    image VARCHAR(255),
    isTopNews TINYINT,
    adminID INT,
    FOREIGN KEY (adminID) REFERENCES Admin(adminID)
);

-- 5. MatchResult Table (Added ON DELETE CASCADE for Fixture reference)
CREATE TABLE MatchResult (
    resultID INT AUTO_INCREMENT PRIMARY KEY,
    homeScore INT,
    awayScore INT,
    fixtureID INT,
    adminID INT,
    FOREIGN KEY (fixtureID) REFERENCES Fixture(fixtureID) ON DELETE CASCADE,
    FOREIGN KEY (adminID) REFERENCES Admin(adminID)
);

-- 6. LeagueTable Table (Added ON DELETE CASCADE for Team and MatchResult references)
CREATE TABLE LeagueTable (
    tableID INT AUTO_INCREMENT PRIMARY KEY,
    played INT,
    won INT,
    lost INT,
    drawn INT,
    gf INT,
    ga INT,
    gd INT,
    points INT,
    resultID INT,
    teamID INT,
    FOREIGN KEY (resultID) REFERENCES MatchResult(resultID) ON DELETE CASCADE,
    FOREIGN KEY (teamID) REFERENCES Team(teamID) ON DELETE CASCADE
);
